1//! The spider trait and request bootstrap types.
2//!
3//! [`Spider`] is the main contract every crawler implements. It defines how
4//! a crawl starts and how each downloaded response turns into scraped items and
5//! follow-up requests.
6//!
7//! ## Example
8//!
9//! ```rust,ignore
10//! use spider_core::{ParseContext, Spider};
11//! use spider_util::{error::SpiderError, item::ParseOutput};
12//! use async_trait::async_trait;
13//!
14//! #[spider_macro::scraped_item]
15//! struct Article {
16//!     title: String,
17//!     content: String,
18//! }
19//!
20//! // State for tracking page count
21//! use std::sync::Arc;
22//! use std::sync::atomic::{AtomicUsize, Ordering};
23//! use dashmap::DashMap;
24//!
25//! #[derive(Clone, Default)]
26//! struct ArticleSpiderState {
27//!     page_count: Arc<AtomicUsize>,
28//!     visited_urls: Arc<DashMap<String, bool>>,
29//! }
30//!
31//! impl ArticleSpiderState {
32//!     fn increment_page_count(&self) {
33//!         self.page_count.fetch_add(1, Ordering::SeqCst);
34//!     }
35//!
36//!     fn mark_url_visited(&self, url: String) {
37//!         self.visited_urls.insert(url, true);
38//!     }
39//! }
40//!
41//! struct ArticleSpider;
42//!
43//! #[async_trait]
44//! impl Spider for ArticleSpider {
45//!     type Item = Article;
46//!     type State = ArticleSpiderState;
47//!
48//!     fn start_requests(&self) -> Result<StartRequests<'_>, SpiderError> {
49//!         let req = Request::new("https://example.com/articles".parse()?);
50//!         Ok(StartRequests::iter(std::iter::once(Ok(req))))
51//!     }
52//!
53//!     async fn parse(&self, cx: ParseContext<'_, Self>) -> Result<(), SpiderError> {
54//!         // Update state - can be done concurrently without blocking the spider
55//!         cx.state().increment_page_count();
56//!         cx.state().mark_url_visited(cx.url.to_string());
57//!
58//!         // Extract articles from the page
59//!         // ... parsing logic ...
60//!
61//!         // Add discovered articles to output
62//!         // cx.add_item(Article { title, content }).await?;
63//!
64//!         // Add new URLs to follow
65//!         // cx.add_request(new_request).await?;
66//!
67//!         Ok(())
68//!     }
69//! }
70//! ```

85/// A boxed iterator of start requests.
86pub type StartRequestIter<'a> = Box<dyn Iterator<Item = Result<Request, SpiderError>> + Send + 'a>;
87
88/// Initial request source returned by [`Spider::start_requests`].
89///
90/// Use [`StartRequests::Urls`] for simple static seeds, [`StartRequests::Iter`]
91/// when you need to construct full [`Request`] values or generate seeds
92/// lazily, and [`StartRequests::File`] when you want to keep large seed lists
93/// outside compiled code.
94pub enum StartRequests<'a> {
95    /// Fixed list of seed URLs.
96    Urls(Vec<&'a str>),
97    /// Direct request iterator supplied by the spider.
98    Iter(StartRequestIter<'a>),
99    /// Path to a plain-text seed file (one URL per line).
100    File(&'a str),
101}
102
103impl<'a> StartRequests<'a> {
104    /// Creates an iterator-based source from any compatible request iterator.
105    pub fn iter<I>(iter: I) -> Self
106    where
107        I: Iterator<Item = Result<Request, SpiderError>> + Send + 'a,
108    {
109        StartRequests::Iter(Box::new(iter))
110    }
111
112    /// Creates a file-based source from a path string.
113    ///
114    /// The file is expected to contain one URL per line. Empty lines and lines
115    /// starting with `#` are ignored.
116    pub fn file(path: &'a str) -> Self {
117        StartRequests::File(path)
118    }
119
120    /// Resolves this source into a concrete request iterator.
121    #[allow(clippy::should_implement_trait)]
122    ///
123    /// URL strings are parsed eagerly as the iterator is consumed. Invalid file
124    /// entries become `SpiderError::ConfigurationError` items that preserve the
125    /// original line number.
126    pub fn into_iter(self) -> Result<StartRequestIter<'a>, SpiderError> {
127        match self {
128            StartRequests::Urls(urls) => {
129                let requests = urls
130                    .into_iter()
131                    .map(|u| Url::parse(u).map(Request::new).map_err(SpiderError::from));
132                Ok(Box::new(requests))
133            }
134            StartRequests::Iter(iter) => Ok(iter),
135            StartRequests::File(path) => start_requests_from_file(path),
136        }
137    }
138}
139
140impl<'a, I> From<I> for StartRequests<'a>
141where
142    I: Iterator<Item = Result<Request, SpiderError>> + Send + 'a,
143{
144    fn from(iter: I) -> Self {
145        StartRequests::iter(iter)
146    }
147}
148
149fn start_requests_from_file<P: AsRef<Path>>(
150    path: P,
151) -> Result<StartRequestIter<'static>, SpiderError> {
152    let path = path.as_ref();
153    let file = File::open(path)?;
154    let path_display = path.display().to_string();
155    let mut lines = BufReader::new(file).lines().enumerate();
156
157    let iter = std::iter::from_fn(move || {
158        loop {
159            let (line_idx, line_res) = lines.next()?;
160            let line_number = line_idx + 1;
161            match line_res {
162                Ok(line) => {
163                    let trimmed = line.trim();
164                    if trimmed.is_empty() || trimmed.starts_with('#') {
165                        continue;
166                    }
167
168                    return Some(match Url::parse(trimmed) {
169                        Ok(url) => Ok(Request::new(url)),
170                        Err(e) => Err(SpiderError::ConfigurationError(format!(
171                            "Invalid start URL in {} at line {}: {}",
172                            path_display, line_number, e
173                        ))),
174                    });
175                }
176                Err(e) => {
177                    return Some(Err(SpiderError::IoError(format!(
178                        "Failed reading {} at line {}: {}",
179                        path_display, line_number, e
180                    ))));
181                }
182            }
183        }
184    });
185
186    Ok(Box::new(iter))
187}

189/// Parse-time context passed into [`Spider::parse`].
190///
191/// This bundles the current [`Response`], shared spider state, and the async
192/// output sink into a single value so user-facing parse signatures stay small.
193///
194/// The context dereferences to [`Response`], which means selector-heavy code
195/// can keep the natural `cx.css(...)` style without manually reaching through a
196/// nested response field.
197pub struct ParseContext<'a, S: Spider + ?Sized> {
198    response: Response,
199    state: &'a S::State,
200    output: ParseOutput<S::Item>,
201}
202
203impl<'a, S: Spider + ?Sized> ParseContext<'a, S> {
204    pub(crate) fn new(
205        response: Response,
206        state: &'a S::State,
207        output: ParseOutput<S::Item>,
208    ) -> Self {
209        Self {
210            response,
211            state,
212            output,
213        }
214    }
215
216    /// Returns the shared spider state for this parse call.
217    pub fn state(&self) -> &'a S::State {
218        self.state
219    }
220
221    /// Returns the current response explicitly.
222    pub fn response(&self) -> &Response {
223        &self.response
224    }
225
226    /// Returns the current response as a mutable reference.
227    pub fn response_mut(&mut self) -> &mut Response {
228        &mut self.response
229    }
230
231    /// Returns the underlying async parse output sink.
232    pub fn output(&self) -> &ParseOutput<S::Item> {
233        &self.output
234    }
235
236    /// Emits a scraped item into the runtime.
237    pub async fn add_item(&self, item: S::Item) -> Result<(), SpiderError> {
238        self.output.add_item(item).await
239    }
240
241    /// Emits multiple scraped items into the runtime.
242    pub async fn add_items(
243        &self,
244        items: impl IntoIterator<Item = S::Item>,
245    ) -> Result<(), SpiderError> {
246        self.output.add_items(items).await
247    }
248
249    /// Emits a follow-up request into the runtime.
250    pub async fn add_request(&self, request: Request) -> Result<(), SpiderError> {
251        self.output.add_request(request).await
252    }
253
254    /// Emits multiple follow-up requests into the runtime.
255    pub async fn add_requests(
256        &self,
257        requests: impl IntoIterator<Item = Request>,
258    ) -> Result<(), SpiderError> {
259        self.output.add_requests(requests).await
260    }
261
262    /// Consumes the context and returns the inner response, state reference,
263    /// and output sink.
264    pub fn into_parts(self) -> (Response, &'a S::State, ParseOutput<S::Item>) {
265        (self.response, self.state, self.output)
266    }
267
268    /// Consumes the context and returns the inner response.
269    pub fn into_response(self) -> Response {
270        self.response
271    }
272}
273
274impl<S: Spider + ?Sized> Deref for ParseContext<'_, S> {
275    type Target = Response;
276
277    fn deref(&self) -> &Self::Target {
278        &self.response
279    }
280}
281
282impl<S: Spider + ?Sized> DerefMut for ParseContext<'_, S> {
283    fn deref_mut(&mut self) -> &mut Self::Target {
284        &mut self.response
285    }
286}

288/// Defines the contract for a spider.
289///
290/// ## Type Parameters
291///
292/// - `Item`: The type of scraped data structure (must implement [`ScrapedItem`])
293/// - `State`: The type of shared state (must implement `Default`)
294///
295/// ## Design Notes
296///
297/// The trait uses `&self` (immutable reference) instead of `&mut self` for the
298/// [`parse`](Spider::parse) method. This design enables efficient concurrent crawling
299/// by eliminating the need for mutex locks when accessing the spider from multiple
300/// async tasks. State that needs mutation should be stored in the associated
301/// `State` type using thread-safe primitives like `Arc<AtomicUsize>` or `DashMap`.
302///
303/// A typical crawl lifecycle looks like this:
304///
305/// 1. [`start_requests`](Spider::start_requests) produces the initial requests
306/// 2. the runtime schedules and downloads them
307/// 3. [`parse`](Spider::parse) receives a [`ParseContext`] for each response
308/// 4. emitted items go to pipelines and emitted requests go back to the scheduler
309#[async_trait]
310pub trait Spider: Send + Sync + 'static {
311    /// The type of item that the spider scrapes.
312    ///
313    /// This associated type must implement the [`ScrapedItem`] trait, which
314    /// provides methods for type erasure, cloning, and JSON serialization.
315    /// Use the `#[scraped_item]` procedural macro to automatically implement
316    /// all required traits for your data structures.
317    type Item: ScrapedItem;
318
319    /// The type of state that the spider uses.
320    ///
321    /// The state type must implement `Default` so it can be instantiated
322    /// automatically by the crawler. It should also be `Send + Sync` to
323    /// enable safe concurrent access from multiple async tasks.
324    ///
325    /// ## Example
326    ///
327    /// ```rust,ignore
328    /// use std::sync::Arc;
329    /// use std::sync::atomic::{AtomicUsize, Ordering};
330    /// use dashmap::DashMap;
331    ///
332    /// #[derive(Clone, Default)]
333    /// struct MySpiderState {
334    ///     page_count: Arc<AtomicUsize>,
335    ///     visited_urls: Arc<DashMap<String, bool>>,
336    /// }
337    /// ```
338    type State: Default + Send + Sync;
339
340    /// Returns static seed URLs.
341    ///
342    /// This method is optional and useful for simple spiders. The default
343    /// [`start_requests`](Spider::start_requests) implementation converts these
344    /// URLs into a request iterator.
345    ///
346    /// Prefer this method when plain URL strings are enough. Override
347    /// [`start_requests`](Spider::start_requests) instead when you need custom
348    /// headers, methods, request metadata, seed-file loading, or dynamic seed
349    /// generation.
350    fn start_urls(&self) -> Vec<&'static str> {
351        Vec::new()
352    }
353
354    /// Returns the initial request source used to start crawling.
355    ///
356    /// The default implementation converts [`start_urls`](Spider::start_urls)
357    /// into an iterator.
358    ///
359    /// To load from seed file, return `StartRequests::file(path)`.
360    /// To use a fixed list of URL strings, return `StartRequests::Urls(...)`.
361    /// To use custom generation logic, return `StartRequests::iter(...)`.
362    ///
363    /// This method is the better override point whenever initial requests need
364    /// more than a URL string, such as per-request metadata, POST bodies, or
365    /// custom headers.
366    ///
367    /// ## Example
368    ///
369    /// ```rust,ignore
370    /// # use spider_core::{scraped_item, ParseContext, Spider, StartRequests};
371    /// # use spider_util::{error::SpiderError, item::{ParseOutput, ScrapedItem}};
372    /// # #[scraped_item]
373    /// # struct ExampleItem {
374    /// #     value: String,
375    /// # }
376    /// # struct MySpider;
377    /// # #[async_trait::async_trait]
378    /// # impl Spider for MySpider {
379    /// #     type Item = ExampleItem;
380    /// #     type State = ();
381    /// fn start_requests(&self) -> Result<StartRequests<'_>, SpiderError> {
382    ///     Ok(StartRequests::file("seeds/start_urls.txt"))
383    /// }
384    /// # async fn parse(&self, cx: ParseContext<'_, Self>) -> Result<(), SpiderError> {
385    /// #     todo!()
386    /// # }
387    /// # }
388    /// ```
389    fn start_requests(&self) -> Result<StartRequests<'_>, SpiderError> {
390        Ok(StartRequests::Urls(self.start_urls()))
391    }
392
393    /// Parses a response and extracts scraped items and new requests.
394    ///
395    /// # Errors
396    ///
397    /// This is the primary method where scraping logic is implemented. It receives
398    /// a [`Response`] object and should extract structured data (items) and/or
399    /// discover new URLs to crawl (requests).
400    ///
401    /// ## Parameters
402    ///
403    /// - `cx`: A parse context containing the current response, shared spider
404    ///   state, and async output sink
405    ///
406    /// ## Returns
407    ///
408    /// The provided [`ParseContext`] lets the spider stream:
409    /// - Scraped items of type `Self::Item`
410    /// - New [`Request`] objects to be enqueued
411    ///
412    /// The usual pattern is:
413    /// - read the response through the context directly, for example
414    ///   `cx.css(...)` via [`Deref`]
415    /// - read shared state with [`ParseContext::state`]
416    /// - call [`ParseContext::add_item`] or `add_items` for scraped items
417    /// - call [`ParseContext::add_request`] or `add_requests` for follow-up
418    ///   requests
419    /// - return `Ok(())` when parsing is done
420    ///
421    /// ## Design Notes
422    ///
423    /// This method takes an immutable reference to `self` (`&self`) instead of
424    /// mutable (`&mut self`), eliminating the need for mutex locks when accessing
425    /// the spider in concurrent environments. State that needs to be modified
426    /// should be stored in the `State` type using thread-safe primitives.
427    ///
428    /// # Errors
429    ///
430    /// Returns a [`SpiderError`] if parsing fails or if an unrecoverable error
431    /// occurs during processing.
432    ///
433    /// # Example
434    ///
435    /// ```rust,ignore
436    /// # use spider_core::{scraped_item, ParseContext, Spider, StartRequests};
437    /// # use spider_util::{error::SpiderError, item::{ParseOutput, ScrapedItem}};
438    /// # use async_trait::async_trait;
439    /// # struct MySpider;
440    /// # #[scraped_item]
441    /// # struct ExampleItem {
442    /// #     value: String,
443    /// # }
444    /// # #[derive(Default)]
445    /// # struct MySpiderState;
446    /// # #[async_trait]
447    /// # impl Spider for MySpider {
448    /// #     type Item = ExampleItem;
449    /// #     type State = MySpiderState;
450    /// #     fn start_requests(&self) -> Result<StartRequests<'_>, SpiderError> {
451    /// #         Ok(StartRequests::iter(std::iter::empty()))
452    /// #     }
453    /// async fn parse(&self, cx: ParseContext<'_, Self>) -> Result<(), SpiderError> {
454    ///     // Parse HTML and extract data
455    ///     let heading = cx.css("h1::text")?.get().unwrap_or_default();
456    ///
457    ///     Ok(())
458    /// }
459    /// # }
460    /// ```
461    async fn parse(&self, cx: ParseContext<'_, Self>) -> Result<(), SpiderError>;
462}
