1//! Middleware Manager for efficient concurrent access.
2//!
3//! This module provides a `MiddlewareManager` that stores middlewares in a vector
4//! and allows efficient concurrent access.
5
6use spider_middleware::middleware::{Middleware, MiddlewareAction};
7use spider_util::error::SpiderError;
8use spider_util::request::Request;
9use spider_util::response::Response;
10use std::sync::Arc;
11
12/// A manager for middlewares that provides efficient concurrent access.
13pub struct MiddlewareManager<C> {
14    middlewares: Vec<Box<dyn Middleware<C> + Send + Sync>>,
15}
16
17impl<C: Send + Sync + 'static> MiddlewareManager<C> {
18    /// Creates a new `MiddlewareManager` with the given middlewares.
19    pub fn new(middlewares: Vec<Box<dyn Middleware<C> + Send + Sync>>) -> Self {
20        Self { middlewares }
21    }
22
23    /// Processes a request through all registered middlewares.
24    pub async fn process_request(
25        &self,
26        client: &C,
27        request: Request,
28    ) -> Result<MiddlewareAction<Request>, SpiderError> {
29        let mut current_request = request;
30
31        for middleware in &self.middlewares {
32            match middleware.process_request(client, current_request).await {
33                Ok(MiddlewareAction::Continue(req)) => {
34                    current_request = req;
35                }
36                Ok(action) => return Ok(action),
37                Err(e) => return Err(e),
38            }
39        }
40
41        Ok(MiddlewareAction::Continue(current_request))
42    }
43
44    /// Processes a response through all registered middlewares in reverse order.
45    pub async fn process_response(
46        &self,
47        response: Response,
48    ) -> Result<MiddlewareAction<Response>, SpiderError> {
49        let mut current_response = response;
50
51        // Process in reverse order to match the request processing chain
52        for middleware in self.middlewares.iter().rev() {
53            match middleware.process_response(current_response).await {
54                Ok(MiddlewareAction::Continue(res)) => {
55                    current_response = res;
56                }
57                Ok(action) => return Ok(action),
58                Err(e) => return Err(e),
59            }
60        }
61
62        Ok(MiddlewareAction::Continue(current_response))
63    }
64
65    /// Processes an error through all registered middlewares in reverse order.
66    pub async fn handle_error(
67        &self,
68        request: &Request,
69        error: &SpiderError,
70    ) -> Result<MiddlewareAction<Request>, SpiderError> {
71        let mut current_error = error.clone();
72
73        for middleware in self.middlewares.iter().rev() {
74            match middleware.handle_error(request, &current_error).await {
75                Ok(action) => return Ok(action),
76                Err(next_error) => current_error = next_error,
77            }
78        }
79
80        Err(current_error)
81    }
82}

84/// A shared middleware manager that can be safely accessed concurrently.
85/// Middleware state is expected to use interior mutability when needed, so the
86/// request path stays fully concurrent.
87pub struct SharedMiddlewareManager<C> {
88    manager: Arc<MiddlewareManager<C>>,
89    has_middlewares: bool,
90}
91
92impl<C: Send + Sync + 'static> SharedMiddlewareManager<C> {
93    /// Creates a new `SharedMiddlewareManager` with the given middlewares.
94    pub fn new(middlewares: Vec<Box<dyn Middleware<C> + Send + Sync>>) -> Self {
95        let has_middlewares = !middlewares.is_empty();
96        Self {
97            manager: Arc::new(MiddlewareManager::new(middlewares)),
98            has_middlewares,
99        }
100    }
101
102    /// Processes a request through all registered middlewares.
103    pub async fn process_request(
104        &self,
105        client: &C,
106        request: Request,
107    ) -> Result<MiddlewareAction<Request>, SpiderError> {
108        if !self.has_middlewares {
109            return Ok(MiddlewareAction::Continue(request));
110        }
111        self.manager.process_request(client, request).await
112    }
113
114    /// Processes a response through all registered middlewares in reverse order.
115    pub async fn process_response(
116        &self,
117        response: Response,
118    ) -> Result<MiddlewareAction<Response>, SpiderError> {
119        if !self.has_middlewares {
120            return Ok(MiddlewareAction::Continue(response));
121        }
122        self.manager.process_response(response).await
123    }
124
125    /// Processes a downloader error through all registered middlewares in reverse order.
126    pub async fn handle_error(
127        &self,
128        request: &Request,
129        error: &SpiderError,
130    ) -> Result<MiddlewareAction<Request>, SpiderError> {
131        if !self.has_middlewares {
132            return Err(error.clone());
133        }
134        self.manager.handle_error(request, error).await
135    }
136}
137
138impl<C: Send + Sync + 'static> Clone for SharedMiddlewareManager<C> {
139    fn clone(&self) -> Self {
140        Self {
141            manager: Arc::clone(&self.manager),
142            has_middlewares: self.has_middlewares,
143        }
144    }
145}
